\section{3.2. Mô hình hồi quy biến chỉ số (M1)}

% --- Slide 1: 3.2.1. Thiết lập mô hình ---
\begin{frame}{3.2.1. Thiết lập mô hình}
    \begin{itemize}
        \item \textbf{Phương pháp:} Sử dụng hồi quy tuyến tính bội với các biến giả (dummy variables) để mô hình hóa thành phần mùa vụ.
        \item \textbf{Các tập biến giải thích:}
        \begin{itemize}
            \item \textbf{23 biến chỉ số giờ:} Đại diện cho 24 giờ trong ngày (lấy giờ 0 làm cơ sở).
            \item \textbf{6 biến chỉ số ngày:} Đại diện cho các ngày trong tuần (lấy Chủ nhật làm cơ sở).
        \end{itemize}
        \item \textbf{Thông số kỹ thuật:}
        \begin{itemize}
            \item \textbf{Số tham số:} 30 tham số (1 hằng số chặn + 23 biến giờ + 6 biến ngày).
            \item \textbf{Ước lượng:} Sử dụng phương pháp Bình phương tối thiểu thông thường (OLS).
        \end{itemize}
    \end{itemize}
    \begin{block}{Cấu trúc toán học của mô hình}
        $$Y_t = \beta_0 + \sum_{i=1}^{23} \beta_i D_{i,t} + \sum_{j=1}^{6} \gamma_j W_{j,t} + \epsilon_t \quad (2)$$
    \end{block}
\end{frame}

\section{3.3. Mô hình hồi quy điều hòa Fourier (M2)}

% --- Slide 3.3.1: Thiết lập ---
\begin{frame}{3.3.1. Thiết lập mô hình}
    \begin{itemize}
        \item \textbf{Ý tưởng:} Thay 23 biến giả giờ của M1 bằng $K$ cặp sóng hài (sin/cos) với chu kỳ $m=24$.
        \item \textbf{Ưu điểm:} Số tham số tối thiểu, mô tả mượt biên dạng mùa vụ.
        \item \textbf{Lựa chọn $K$:} Tối ưu theo tiêu chí AIC trên tập huấn luyện với $K \in \{1, 2, 3, 4\}$.
    \end{itemize}
    \begin{block}{Cấu trúc toán học}
        $$Y_t = \alpha_0 + \sum_{k=1}^{K}\left[a_k \cos\!\left(\frac{2\pi k t}{m}\right) + b_k \sin\!\left(\frac{2\pi k t}{m}\right)\right] + \varepsilon_t$$
        \centering Số tham số: $2K + 1$
    \end{block}
\end{frame}

% --- Slide 3.3.2: Lựa chọn K ---
\begin{frame}{3.3.2. Lựa chọn số sóng hài $K^*$}
    \begin{columns}[T]
        \begin{column}{0.5\textwidth}
            \begin{itemize}
                \item $K=2$ cho AIC \textbf{thấp nhất}: $30.769,23$.
                \item Thêm sóng từ $K \ge 3$ không cải thiện đáng kể nhưng tăng độ phức tạp.
                \item \textbf{Chọn:} $K^* = 2$ (5 tham số).
            \end{itemize}
        \end{column}

\end{frame}

\begin{frame}{3.3.4. Chẩn đoán phần dư}
    \begin{columns}[T]
        \begin{column}{0.45\textwidth}
            \begin{itemize}
                \item \textbf{Ljung--Box Q(48):} $p < 0,001$ → bác bỏ nhiễu trắng.
                \item \textbf{ACF:} Còn spike rõ tại chu kỳ 24h, 48h.

            \end{itemize}
        \end{column}

\section{3.4. Mô hình SARIMA (M3)}

% --- Slide 3.4.1: Nhận dạng ---
\begin{frame}{3.4.1. Nhận dạng mô hình}
    \begin{itemize}
        \item \textbf{Cơ sở từ EDA:} Chuỗi dừng nhưng có cấu trúc mùa $s = 24$ \textbf{cực mạnh}.
        \item \textbf{Tiền xử lý:} Áp dụng sai phân mùa $\nabla_{24}y_t = y_t - y_{t-24}$ ($D = 1$).
        \item \textbf{Quy trình lựa chọn:}
        \begin{itemize}
            \item Hàm \texttt{auto.arima()} với ràng buộc $p, q \le 2$ và $P, Q \le 1$.
            \item Đối chiếu chéo với phân tích ACF/PACF của chuỗi sau sai phân.
            \item So sánh các ứng viên thủ công theo AIC.
        \end{itemize}
    \end{itemize}
    \begin{block}{Mô hình được chọn}
        \centering
        \large $\text{SARIMA}(2, 0, 2)(1, 1, 0)_{24}$
    \end{block}
\end{frame}

% --- Slide 3.4.2: Cấu trúc toán học ---
\begin{frame}{3.4.2. Cấu trúc toán học}
    \begin{block}{Phương trình SARIMA$(2,0,2)(1,1,0)_{24}$}
        \small
        $$(1 - \phi_1 B - \phi_2 B^2)(1 - \Phi_1 B^{24})\,\nabla_{24}\,y_t = (1 + \theta_1 B + \theta_2 B^2)\,\varepsilon_t$$
        \centering $\varepsilon_t \sim \mathcal{N}(0, \sigma^2)$
    \end{block}
    \textbf{Diễn giải các thành phần:}
    \begin{itemize}
        \item \textbf{AR(2):} $\phi_1, \phi_2$ -- phụ thuộc vào 2 giờ liền trước.
        \item \textbf{MA(2):} $\theta_1, \theta_2$ -- hiệu chỉnh nhiễu của 2 bước trước.
        \item \textbf{SAR(1)$_{24}$:} $\Phi_1$ -- phụ thuộc vào \emph{cùng giờ} của ngày trước.
        \item \textbf{Sai phân mùa $D = 1$:} loại bỏ chu kỳ ngày $\Rightarrow$ chuỗi dừng.
    \end{itemize}
\end{frame}

% --- Slide 3.4.3: Kết quả train/test ---
\begin{frame}{3.4.3. Kết quả ước lượng và dự báo}
    \begin{columns}[T]
        \begin{column}{0.5\textwidth}
            \begin{block}{Train -- Xuất sắc}
                \begin{itemize}
                    \item RMSE $= 154,21$ MW.
                    \item MAPE $= \mathbf{0,80\%}$.
                    \item AIC $= 22.540$ (chuỗi sai phân$^*$).
                \end{itemize}
            \end{block}
            \begin{alertblock}{Test -- Suy giảm nghiêm trọng}
                \begin{itemize}
                    \item RMSE $= 2.676,65$ MW.
                    \item MAPE $= \mathbf{13,93\%}$.
                \end{itemize}
            \end{alertblock}
            \footnotesize $^*$ AIC không so sánh trực tiếp với M1, M2.
        \end{column}
        \begin{column}{0.5\textwidth}
            \begin{figure}
                \centering
                \includegraphics[width=\textwidth]{img/fig_s4_07_model3_fit.png}
                \caption{Dự báo M3 SARIMA trên tập kiểm tra}
            \end{figure}
        \end{column}
    \end{columns}
\end{frame}

% --- Slide 3.4.4: Phân tích Overfitting (slide nhấn mạnh) ---
\begin{frame}{3.4.4. Phân tích hiện tượng quá khớp (Overfitting)}
    \begin{block}{Quan sát chính}
        Tỷ số RMSE Test / Train $\approx \mathbf{17}$ lần $\Rightarrow$ Dấu hiệu overfitting điển hình.
    \end{block}
    \textbf{Nguyên nhân gốc rễ:}
    \begin{enumerate}
        \item \textbf{Dữ liệu huấn luyện ngắn:} Chỉ 74 ngày -- chưa đủ để ổn định ARMA(2,2).
        \item \textbf{Mô hình ghi nhớ nhiễu:} Bắt cấu trúc nhiễu ngắn hạn trong train thay vì học cấu trúc mùa vụ tổng quát.
        \item \textbf{Phân phối khác biệt:} Đặc điểm tiêu thụ mùa hè sớm (train) khác mùa hè muộn (test).
    \end{enumerate}
    \textbf{Bài học:} SARIMA mạnh ở dự báo \emph{ngắn hạn} (24--48h) nhưng cần dữ liệu \emph{đủ dài} để ổn định.
\end{frame}

% --- Slide 3.4.5: Chẩn đoán phần dư ---
\begin{frame}{3.4.5. Chẩn đoán phần dư}
    \begin{columns}[T]
        \begin{column}{0.45\textwidth}
            \begin{itemize}
                \item \textbf{Miền thời gian:} Phần dư trông \emph{gần nhiễu trắng}.
                \item \textbf{ACF:} Giảm rất nhanh, phù hợp cấu trúc ARMA.
                \item \textbf{Ljung--Box Q(48):} $p < 0,001$ -- vẫn bác bỏ nhiễu trắng (cấu trúc nhỏ chưa nắm hết).
                \item \textbf{Q-Q plot:} Phân phối phần dư \textbf{gần chuẩn hơn} M1, M2.
            \end{itemize}
        \end{column}
        \begin{column}{0.55\textwidth}
